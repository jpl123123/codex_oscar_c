cmake_minimum_required(VERSION 3.20)
project(oscar_ascend_ops LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "OSCAR AscendC must be compiled on Linux with CANN and torch_npu; no CPU substitute is built.")
endif()
if(NOT ASCEND_HOME_PATH)
  set(ASCEND_HOME_PATH "$ENV{ASCEND_HOME_PATH}")
endif()
if(NOT SOC_VERSION)
  message(FATAL_ERROR "Set -DSOC_VERSION to the actual supported Ascend 910B/910_93 SoC.")
endif()
if(NOT SOC_VERSION MATCHES "[Aa]scend910[Bb_].*")
  message(FATAL_ERROR "This mixed 1C2V implementation targets the 910B/910_93 architecture; other SoCs are unvalidated.")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter Development)
find_package(Threads REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import torch; print(torch.utils.cmake_prefix_path)"
  OUTPUT_VARIABLE TORCH_CMAKE_PREFIX OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE TORCH_IMPORT_RESULT)
if(NOT TORCH_IMPORT_RESULT EQUAL 0)
  message(FATAL_ERROR "Selected Python cannot import torch")
endif()
list(APPEND CMAKE_PREFIX_PATH "${TORCH_CMAKE_PREFIX}")
find_package(Torch REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" -c "import pathlib,torch_npu; print(pathlib.Path(torch_npu.__file__).parent)"
  OUTPUT_VARIABLE TORCH_NPU_PATH OUTPUT_STRIP_TRAILING_WHITESPACE RESULT_VARIABLE TORCH_NPU_IMPORT_RESULT)
if(NOT TORCH_NPU_IMPORT_RESULT EQUAL 0)
  message(FATAL_ERROR "Selected Python cannot import torch_npu")
endif()
set(ASCEND_CANN_PACKAGE_PATH "${ASCEND_HOME_PATH}")
set(RUN_MODE "npu")
foreach(REL tools/tikcpp/ascendc_kernel_cmake compiler/tikcpp/ascendc_kernel_cmake ascendc_devkit/tikcpp/samples/cmake)
  if(EXISTS "${ASCEND_HOME_PATH}/${REL}/ascendc.cmake")
    set(ASCENDC_CMAKE_DIR "${ASCEND_HOME_PATH}/${REL}")
    break()
  endif()
endforeach()
if(NOT ASCENDC_CMAKE_DIR)
  message(FATAL_ERROR "No ascendc.cmake under ASCEND_HOME_PATH=${ASCEND_HOME_PATH}")
endif()
include("${ASCENDC_CMAKE_DIR}/ascendc.cmake")
include_directories("${CMAKE_CURRENT_SOURCE_DIR}/include" "${ASCEND_HOME_PATH}/include")
ascendc_library(oscar_ascend_kernels SHARED op_kernel/oscar_vector.cpp op_kernel/oscar_history_cv.cpp op_kernel/oscar_window.cpp op_kernel/oscar_prefix.cpp op_kernel/oscar_calibration.cpp)
add_library(oscar_ascend_ops SHARED torch/bindings.cpp torch/calibration_bindings.cpp op_host/tiling.cpp)
target_include_directories(oscar_ascend_ops PRIVATE "${TORCH_NPU_PATH}/include" "${TORCH_NPU_PATH}/.." ${TORCH_INCLUDE_DIRS})
target_link_directories(oscar_ascend_ops PRIVATE "${TORCH_NPU_PATH}/lib" "${ASCEND_HOME_PATH}/lib64")
target_link_libraries(oscar_ascend_ops PRIVATE oscar_ascend_kernels ${TORCH_LIBRARIES} torch_npu ascendcl tiling_api platform register Threads::Threads)
set_target_properties(oscar_ascend_ops PROPERTIES BUILD_RPATH "$ORIGIN;${TORCH_NPU_PATH}/lib;${ASCEND_HOME_PATH}/lib64")
separate_arguments(TORCH_FLAGS NATIVE_COMMAND "${TORCH_CXX_FLAGS}")
target_compile_options(oscar_ascend_ops PRIVATE ${TORCH_FLAGS})
